# 构建系统架构


### 1.1 顶层调用链

```
make world
 └─ target/stamp-install                (toplevel.mk:133)
      └─ target/linux/install           (target/Makefile:29, builddirs-install=linux)
           └─ target/linux/mediatek/install  (target/linux/Makefile:11)
                └─ BuildTarget = BuildKernel  (include/target.mk:391 → include/kernel-build.mk)
                     ├─ compile: $(LINUX_DIR)/.modules
                     │    └─ +make -C image compile TARGET_BUILD=   (kernel-build.mk:157)
                     │         └─ image/Makefile → $(eval $(call BuildImage))
                     │              ├─ Device/Build/kernel (DTS 编译 + KERNEL 管道)
                     │              └─ Device/Build/image (rootfs 编译 + 镜像合成)
                     └─ install: $(LINUX_DIR)/.image
                          └─ +make -C image compile install TARGET_BUILD=  (kernel-build.mk:176-177)
```

关键路径：
- **kernel-build.mk:157**: `compile: $(LINUX_DIR)/.modules` → `+$(MAKE) -C image compile TARGET_BUILD=`
- **kernel-build.mk:176**: `install: $(LINUX_DIR)/.image` → `+$(MAKE) -C image compile install TARGET_BUILD=`

这意味着内核镜像编译和目标镜像编译都通过 `target/linux/mediatek/image/Makefile` 执行。`TARGET_BUILD=1` 让 include/image.mk 走 `Device/Build/*` 分支。

### 1.2 镜像构建的核心输入变量

`target/linux/mediatek/Makefile`（第 1-17 行）：
```makefile
ARCH:=arm                          # 实际 subtarget filogic 是 aarch64
BOARD:=mediatek
SUBTARGETS:=filogic mt7622 mt7623 mt7629
KERNEL_PATCHVER:=6.12              # Linux 6.12.y
```

`target/linux/mediatek/filogic/target.mk`（第 1-11 行）：
```makefile
ARCH:=aarch64                      # 真正的架构
SUBTARGET:=filogic
BOARDNAME:=Filogic 8x0 (MT798x)
CPU_TYPE:=cortex-a53
KERNELNAME:=Image dtbs             # 重要：内核输出 "Image"（非 vmlinux）
DEFAULT_PACKAGES += fitblk ...      # fitblk: FIT 块设备驱动，嵌入 rootfs 到 FIT 镜像时需要
```

### 1.3 Image 构建目标入口：`BuildImage`

`include/image.mk:965-1017` 定义了 `BuildImage`，是所有 target/linux/<board>/image/Makefile 的顶层框架：

```makefile
define BuildImage
  download: prepare: compile: compile-dtb: clean: image_prepare:
  compile:
    $(call Build/Compile)           # 编译所有选中的 packages

  kernel_prepare: image_prepare     # image_prepare = compile + compile-dtb + mkdir BIN_DIR
    $(call Image/Build/targz)
    $(call Image/Build/cpiogz)
    $(call Image/BuildKernel)       # 编译内核 (Kernel/CompileImage)
    $(call Image/InstallKernel)     # 安装内核到 $(KDIR)/tmp/

  $(foreach device,$(TARGET_DEVICES),$(call Device,$(device)))
  # 每个 Device 展开 Device/Init → Device/<name> → Device/Build

  install-images: kernel_prepare $(foreach fs,$(filter-out ubifs,$(TARGET_FILESYSTEMS)),$(KDIR)/root.$(fs))
    $(foreach fs,$(TARGET_FILESYSTEMS), $(call Image/Build,$(fs)))

  install: install-images
    $(call Image/Manifest)
endef
```
